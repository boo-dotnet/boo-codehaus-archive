The patch linked to from BOO-1 includes an implementation of decimal literals, in src/Boo.Lang.Compiler/Units/DecimalUnit.cs:

namespace Boo.Lang.Compiler.Units

{

 using System;

 using Boo.Lang.Compiler;

 using Boo.Lang.Compiler.Ast;

 public class DecimalUnit : AbstractAstUnit

 {

 static string _suffix = "dec";

 public override Expression Apply(UnitLiteralExpression node)
 {
 if (node.Suffix != _suffix) return null;
 
 double val = node.Value;
 
 return AstUtil.CreateMethodInvocationExpression(
 node.LexicalInfo,
 AstUtil.CreateReferenceExpression("System.Decimal"),
 new DoubleLiteralExpression(val));
 }

 }

}
