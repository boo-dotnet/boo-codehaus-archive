### Doug H, 26/Mar/06 7:16 AM

The boo file needs to be utf-8 encoded, probably.

Sharpdevelop is using windows encoding by default.

And unfortunately changing the encoding doesn't work either. (right click on text, choose options, then change the encoding)

I had to delete the boo files and create new ones, after changing the global encoding option:

Go to Tools-&gt;options-&gt;text editor-&gt;general and change the default encoding to utf-8.

Or if you have another text editor like JEdit you can use that to open your existing boo files and change the encoding to utf-8.

### Daniel Grunwald, 26/Mar/06 7:46 AM

Maybe the Boo compiler can use this helper class to auto-detect the encoding:

 public static class FileReader

 {

 public static bool IsUnicode(Encoding encoding)
 {
 int codepage = encoding.CodePage;
 // return true if codepage is any UTF codepage
 return codepage == 65001 || codepage == 65000 || codepage == 1200 || codepage == 1201;
 }

 public static string ReadFileContent(string fileName, ref Encoding encoding, Encoding defaultEncoding)

 {

 using (FileStream fs = new FileStream(fileName, FileMode.Open, FileAccess.Read)) {

 using (StreamReader reader = OpenStream(fs, encoding, defaultEncoding)) 
{
 encoding = reader.CurrentEncoding;
 return reader.ReadToEnd();
 }

 }

 }

 public static StreamReader OpenStream(FileStream fs, Encoding suggestedEncoding, Encoding defaultEncoding)

 {

 if (fs.Length &gt; 3) {

 // the autodetection of StreamReader is not capable of detecting the difference

 // between ISO-8859-1 and UTF-8 without BOM.

 int firstByte = fs.ReadByte();

 int secondByte = fs.ReadByte();

 switch ((firstByte &lt;&lt; 8) | secondByte) 
{
 case 0x0000: // either UTF-32 Big Endian or a binary file; use StreamReader
 case 0xfffe: // Unicode BOM (UTF-16 LE or UTF-32 LE)
 case 0xfeff: // UTF-16 BE BOM
 case 0xefbb: // start of UTF-8 BOM
 // StreamReader autodetection works
 fs.Position = 0;
 return new StreamReader(fs);
 default:
 return AutoDetect(fs, (byte)firstByte, (byte)secondByte, defaultEncoding);
 }

 } else {

 if (suggestedEncoding != null) 
{
 return new StreamReader(fs, suggestedEncoding);
 }

 else 
{
 return new StreamReader(fs);
 }

 }

 }

 static StreamReader AutoDetect(FileStream fs, byte firstByte, byte secondByte, Encoding defaultEncoding)

 {

 int max = (int)Math.Min(fs.Length, 500000); // look at max. 500 KB

 const int ASCII = 0;

 const int Error = 1;

 const int UTF8 = 2;

 const int UTF8Sequence = 3;

 int state = ASCII;

 int sequenceLength = 0;

 byte b;

 for (int i = 0; i &lt; max; i++) {

 if (i == 0) 
{
 b = firstByte;
 }

 else if (i == 1) 
{
 b = secondByte;
 }

 else 
{
 b = (byte)fs.ReadByte();
 }

 if (b &lt; 0x80) {

 // normal ASCII character

 if (state == UTF8Sequence) 
{
 state = Error;
 break;
 }

 } else if (b &lt; 0xc0) {

 // 10xxxxxx : continues UTF8 byte sequence

 if (state == UTF8Sequence) {

 --sequenceLength;

 if (sequenceLength &lt; 0) {
 state = Error;
 break;
 } else if (sequenceLength == 0) {
 state = UTF8;
 }

 } else {
 state = Error;
 break;
 }

 } else if (b &gt;= 0xc2 &amp;&amp; b &lt; 0xf5) {

 // beginning of byte sequence

 if (state == UTF8 || state == ASCII) {

 state = UTF8Sequence;

 if (b &lt; 0xe0) 
{
 sequenceLength = 1; // one more byte following
 }

 else if (b &lt; 0xf0) 
{
 sequenceLength = 2; // two more bytes following
 }

 else 
{
 sequenceLength = 3; // three more bytes following
 }

 } else 
{
 state = Error;
 break;
 }

 } else 
{
 // 0xc0, 0xc1, 0xf5 to 0xff are invalid in UTF-8 (see RFC 3629)
 state = Error;
 break;
 }

 }

 fs.Position = 0;

 switch (state) {

 case ASCII:

 case Error:

 // when the file seems to be ASCII or non-UTF8,

 // we read it using the user-specified encoding so it is saved again

 // using that encoding.

 if (IsUnicode(defaultEncoding)) 
{
 // the file is not Unicode, so don't read it using Unicode even if the
 // user has choosen Unicode as the default encoding.
 
 // If we don't do this, SD will end up always adding a Byte Order Mark
 // to ASCII files.
 defaultEncoding = Encoding.Default; // use system encoding instead
 }

 return new StreamReader(fs, defaultEncoding);

 default:

 return new StreamReader(fs);

 }

 }

 }
