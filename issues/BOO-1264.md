## Description

```
interface INumber[of T]:
 def add(a as T, b as T) as void

class MyVector[of T(INumber[of T])]:
 x as T
 public def add(a as MyVector[of T], b as MyVector[of T]) as void: //BCE0149
 x.add(a.x, b.x)
```
