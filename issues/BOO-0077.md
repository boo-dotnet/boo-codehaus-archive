## Description

Adding a function overload in an extended class should not override the overloaded function in the base class.

Expected behaviour: The commented out lines should compile

//-----------------------------

class BaseClass:

 def Test1() as int:

 return 55

 def Test1(i as int) as int:

 return Test1()/i

 def Test2() as int:

 return 33

class ExtendedClass(BaseClass):

 def Test3(i as int) as int:

 return Test1()*i

 def Test2(i as int) as int:

 // BCE0016: Boo.Lang.Compiler.CompilerError: No

 // overload of the method Test2 takes

 // 0 parameter(s).

 //return Test2()/i

 return 22

ebase = ExtendedClass()

x = ebase.Test1()

// BCE0016: Boo.Lang.Compiler.CompilerError: No

// overload of the method Test2 takes

// 0 parameter(s).

//y = ebase.Test2()

y = ebase.Test2(6)

z = ebase.Test1(5)

a = ebase.Test3(2)

b = ebase.Test2(3)

print("$
{x}

, $
{y}

, $
{z}

, $
{a}

, $
{b}

")
