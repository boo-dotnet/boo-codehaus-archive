## Description

This causes booi and booxw to abort:

struct Test:

 public i as int

 def constructor(j as int):

 i = j

 def op_Addition(j as int) as Test:

 i += j

 return self // this is the culprit

ix = Test(0)

ix.op_Addition(1)
