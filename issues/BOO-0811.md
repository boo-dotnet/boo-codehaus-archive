## Description

If two threads enter at the same time the _converterCache throws a Item has already been added. Key in dictionary:

'System.Boolean' Key being added: 'System.Boolean'

static BoolConverter GetBoolConverter(Type type)

{

 BoolConverter converter = (BoolConverter) _converterCache[type];

 if (null == converter)
 {
 converter = CreateBoolConverter(type);
 _converterCache.Add(type, converter);
 }

 return converter;

}
