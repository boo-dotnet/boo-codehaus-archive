One way to fix may be in src/boo.lang.compiler/typesystem/ExternalProperty.cs, change these properties to the code below.

It looks strange because by later steps, getting a property has already been converted to an externalmethod invocation, while setting a property has not.

All nant tests still succeed. But of course a new test is needed for this that accesses a protected property in an externally compiled base class.

 public bool IsPublic

 {

 get

 {

 System.Reflection.MethodInfo setter = _property.GetSetMethod(true);

 if (null != setter)
 {
 return setter.IsPublic;
 }

 return false;

 }

 }

 public bool IsProtected

 {

 get

 {

 System.Reflection.MethodInfo setter = _property.GetSetMethod(true);

 if (null != setter)
 {
 return setter.IsFamily || setter.IsFamilyOrAssembly;
 }

 return false;

 }

 }

 public bool IsInternal

 {

 get

 {

 System.Reflection.MethodInfo setter = _property.GetSetMethod(true);

 if (null != setter)
 {
 return setter.IsAssembly;
 }

 return false;

 }

 }

 public bool IsPrivate

 {

 get

 {

 System.Reflection.MethodInfo setter = _property.GetSetMethod(true);

 if (null != setter)
 {
 return setter.IsPrivate;
 }

 return false;

 }

 }
